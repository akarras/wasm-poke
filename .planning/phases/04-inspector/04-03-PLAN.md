2. **Modify show_wat_panel() to use scroll_to_row:**
When cursor changes (via keyboard or click), use `scroll_to_row`:

```rust
fn show_wat_panel(&mut self, ui: &mut egui::Ui, selection: &mut SelectionState, scroll_to: Option<usize>) {
    ui.label(egui::RichText::new("WAT Instructions").strong());
    ui.separator();

    if self.cached_wat_lines.is_empty() {
        ui.label("No instructions");
        return;
    }

    let row_height = 18.0;
    let line_count = self.cached_wat_lines.len();

    let mut scroll_area = egui::ScrollArea::vertical()
        .id_salt("wat_scroll");

    // Apply scroll_to_row if cursor changed
    if let Some(row) = scroll_to {
        scroll_area = scroll_area.scroll_to_row(row, Some(egui::Align::Center));
    }

    scroll_area.show_rows(ui, row_height, line_count, |ui, row_range| {
        for row_idx in row_range {
            let is_current = row_idx == selection.instruction_cursor;
            let line = &self.cached_wat_lines[row_idx];

            let response = ui.horizontal(|ui| {
                // Paint background highlight BEFORE text if current
                if is_current {
                    let rect = ui.available_rect_before_wrap();
                    ui.painter().rect_filled(
                        rect,
                        0.0,
                        ui.visuals().selection.bg_fill,
                    );
                }
                ui.label(egui::RichText::new(&line.text).monospace());
            });

            // Make entire row clickable
            if response.response.interact(egui::Sense::click()).clicked() {
                selection.instruction_cursor = row_idx;
            }
        }
    });
}
```
